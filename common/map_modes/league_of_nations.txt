		top = {

		}

							}

						set_temp_variable = { blue = 0.7 }

						set_temp_variable = { red = 0.1 }
						set_temp_variable = { blue = 0.7 }
						set_temp_variable = { green = 0.1 }
						set_temp_variable = { alpha = 0.65 }
					}
				}
			}
			thickness = yes
		}

		far_text = country
		near_text = country

		update_daily = yes
	}
