### Ideology Rules ###

	type = joining_rules
	visible = {
		has_democratic_government_trigger = yes
	}

		has_democratic_government_tooltip_trigger = yes
	}
	
	ai_will_do = {
		base = 0
	}
	can_remove = {
		always = no
	}
