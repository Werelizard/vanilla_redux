			if = {

			if = {
				limit = {
					has_idea = vx_london_naval_treaty_cheat
				}
				remove_ideas = vx_london_naval_treaty_cheat
			}

				limit = {
					has_idea = vx_london_naval_treaty_escalator_clause

				remove_ideas = vx_london_naval_treaty_escalator_clause
			}

				limit = {
					NOT = {
						has_global_flag = vx_naval_treaty_abandoned_global_flag

				}
