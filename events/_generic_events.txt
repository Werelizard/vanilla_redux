#### CRISIS EVENTS ####

# Strikes happen at low stability. Desertions with high casualties and low war support. Mutinies with low war support and high 
# surrender progress. Removed the random chance stuff because I hate it.

country_event = {
	id = crisis_strikes.0
	title = crisis_strikes.0.title
	desc = crisis_strikes.0

	is_triggered_only = yes
	trigger = {
		can_have_strikes_crisis_trigger = yes
		NOT = {
			has_idea = vx_crisis_cooldown
		}
	}

	immediate = {
		
	}


	option = {
		# It'll blow off soon
		name = crisis_strikes.0.a
		if = {
			limit = {
				has_capitulated = no
			}
			add_timed_idea = {
				idea = vx_strikes
				days = 270
			}
			custom_effect_tooltip = vx_unlock_decisions_to_handle_strikes_tt
		}
	}
}
